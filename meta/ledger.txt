  id:                   "required, unique, matches ID FORMAT above"
  title:                "required, imperative or noun phrase, sentence case"
  chapter:              "required, e.g. 01-02"
  layer:                "required, 0|1|2|3"
  tier:                 "required for layers 1-2, one of A B C D E F; null for layers 0 and 3"
  track:                "required for layer 3, one of the seven disciplines; null otherwise"
  prerequisites:        "list of ledger IDs. [] only for true axioms."
  introduces_terms:     "terms first DEFINED here. Feeds glossary + linter."
  introduces_symbols:   "symbols first DEFINED here. Must exist in notation.md."
  uses_symbols:         "symbols used but defined elsewhere. Linter checks order."
  handbook_ref:         "section name + page in Handbook 10.6, or null"
  handbook_verified:    "true only if page confirmed against the PDF"
  spec_lines:           "list of {discipline, area, item} from NCEES specs"
  memorize:             "true if NOT in the Handbook and must be known cold"
  memorize_despite_handbook: "true if IN the Handbook but still worth knowing cold (time cost of lookup exceeds benefit)"
  primer_source:        "chapter ID that borrows this concept in a Primer block; null if not borrowed"
  review_questions:     "count authored"
  practice_problems:    "count authored"
  figures:              "list of figure IDs introduced in this concept's chapter"
  status:               "planned | drafted | reviewed | verified"
  notes:                "free-form authoring notes; not student-facing"
